         * 4. AUDIO & SPEECH SYNTHESIS ENGINE (Web Speech & Web Audio API)

        function speakText(text) {
            if ('speechSynthesis' in window) {
                window.speechSynthesis.cancel(); // Stop current speaking
                const utterance = new SpeechSynthesisUtterance(text);
                utterance.lang = 'zh-HK'; // Cantonese / Traditional Chinese
                utterance.rate = 0.85; // Slightly slower for K2 children
                window.speechSynthesis.speak(utterance);
            }
        }

        function playAudioEffect(type) {
            try {
                const ctx = new (window.AudioContext || window.webkitAudioContext)();
                const osc = ctx.createOscillator();
                const gain = ctx.createGain();
                osc.connect(gain);
                gain.connect(ctx.destination);

                if (type === 'click') {
                    osc.frequency.setValueAtTime(400, ctx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.0001, ctx.currentTime + 0.1);
                    osc.start();
                    osc.stop(ctx.currentTime + 0.1);
                } else if (type === 'star') {
                    osc.frequency.setValueAtTime(523.25, ctx.currentTime);
                    osc.frequency.exponentialRampToValueAtTime(880, ctx.currentTime + 0.3);
                    gain.gain.exponentialRampToValueAtTime(0.0001, ctx.currentTime + 0.3);
                    osc.start();
                    osc.stop(ctx.currentTime + 0.3);
                }
            } catch (e) {
                console.log('Audio Context not allowed without interaction');
            }
        }
